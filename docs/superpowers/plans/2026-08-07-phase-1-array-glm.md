# Boldtailor Phase 1 Array GLM Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a tested, array-based conventional first-level GLM that accepts run-wise events and confounds, fits Nilearn OLS or AR models, computes named t contrasts across runs, and returns immutable numerical results with in-sample R-squared diagnostics.

**Architecture:** `AnalysisData` owns normalized `time x features` arrays and copied tabular inputs; `ModelSpec` owns conventional design and contrast configuration; `fit()` compiles run designs with Nilearn, calls the array-level GLM, combines independent run contrasts, and returns `AnalysisResult`. Phase 1 has no BIDS, NIfTI, CIFTI, Stats Models JSON, adaptive HRFs, GLMsingle recipe, or file output.

**Tech Stack:** Python 3.12+, uv, NumPy, pandas, Nilearn 0.13.1+, setuptools, pytest, pytest-cov.

## Global Constraints

- Use uv for package management and `uv run` for every local Python or test command.
- Use pytest and strict RED-GREEN-Refactor development.
- Commit each failing test before writing its implementation.
- Do not alter a failing test merely to make an incomplete implementation pass.
- Keep every `__init__.py` file empty: no code, imports, exports, comments, or docstrings.
- Keep functions short and modules focused on one responsibility.
- Scientific arrays use `time x features` orientation.
- Nilearn is the only Phase 1 GLM backend.
- `fit()` performs no file I/O and does not mutate inputs.
- Phase 1 supports named t contrasts only. F contrasts belong to the later model-compilation phase.
- Run all commits through `uv run --no-project git`.

```python
from __future__ import annotations

from dataclasses import dataclass
from typing import Sequence

import numpy as np
import pandas as pd


_EVENT_COLUMNS = ("onset", "duration", "trial_type")


@dataclass(frozen=True)
class AnalysisData:
    _signals: tuple[np.ndarray, ...]
    _events: tuple[pd.DataFrame, ...]
    _confounds: tuple[pd.DataFrame, ...]
    _frame_times: tuple[np.ndarray, ...]

    @property
    def n_runs(self) -> int:
        return len(self._signals)

    @property
    def n_features(self) -> int:
        return self._signals[0].shape[1]

    @property
    def signals(self) -> tuple[np.ndarray, ...]:
        return self._signals

    @property
    def events(self) -> tuple[pd.DataFrame, ...]:
        return tuple(frame.copy(deep=True) for frame in self._events)

    @property
    def confounds(self) -> tuple[pd.DataFrame, ...]:
        return tuple(frame.copy(deep=True) for frame in self._confounds)

    @property
    def frame_times(self) -> tuple[np.ndarray, ...]:
        return self._frame_times


def from_arrays(
    signals: np.ndarray | Sequence[np.ndarray],
    events: pd.DataFrame | Sequence[pd.DataFrame],
    *,
    tr: float,
    confounds: pd.DataFrame | Sequence[pd.DataFrame] | None = None,
) -> AnalysisData:
    run_signals = _as_signal_runs(signals)
    run_events = _as_table_runs(events)
    _validate_tr(tr)
    _validate_run_count(run_signals, run_events, "events")

    prepared_signals = tuple(_prepare_signal(run, index) for index, run in enumerate(run_signals))
    _validate_feature_counts(prepared_signals)
    prepared_events = tuple(
        _prepare_events(frame, prepared_signals[index].shape[0], tr, index)
        for index, frame in enumerate(run_events)
    )
    prepared_confounds = _prepare_confounds(confounds, prepared_signals)
    frame_times = tuple(
        _readonly(np.arange(run.shape[0], dtype=float) * tr)
        for run in prepared_signals
    )
    return AnalysisData(
        _signals=prepared_signals,
        _events=prepared_events,
        _confounds=prepared_confounds,
        _frame_times=frame_times,
    )


def _as_signal_runs(value: np.ndarray | Sequence[np.ndarray]) -> tuple[np.ndarray, ...]:
    if isinstance(value, np.ndarray):
        return (value,)
    runs = tuple(value)
    if not runs:
        raise ValueError("signals must contain at least one run")
    return runs


def _as_table_runs(
    value: pd.DataFrame | Sequence[pd.DataFrame],
) -> tuple[pd.DataFrame, ...]:
    if isinstance(value, pd.DataFrame):
        return (value,)
    return tuple(value)


def _validate_tr(tr: float) -> None:
    if not np.isfinite(tr) or tr <= 0:
        raise ValueError("TR must be positive and finite")


def _validate_run_count(
    signals: Sequence[object], tables: Sequence[object], name: str
) -> None:
    if len(signals) != len(tables):
        raise ValueError(f"{name} must contain one table per signal run")


def _prepare_signal(values: np.ndarray, run: int) -> np.ndarray:
    array = np.asarray(values)
    if array.ndim != 2:
        raise ValueError(f"run {run} signals must have shape time x features")
    if array.shape[0] == 0 or array.shape[1] == 0:
        raise ValueError(f"run {run} signals must have nonzero dimensions")
    if not np.isfinite(array).all():
        raise ValueError(f"run {run} signals must be finite")
    return _readonly(array, dtype=float)


def _validate_feature_counts(signals: tuple[np.ndarray, ...]) -> None:
    counts = {run.shape[1] for run in signals}
    if len(counts) != 1:
        raise ValueError("all runs must have the same number of features")


def _prepare_events(
    frame: pd.DataFrame, n_timepoints: int, tr: float, run: int
) -> pd.DataFrame:
    missing = [name for name in _EVENT_COLUMNS if name not in frame]
    if missing:
        raise ValueError(f"run {run} events missing columns: {', '.join(missing)}")
    copied = frame.copy(deep=True).reset_index(drop=True)
    onset = copied["onset"].to_numpy(dtype=float)
    duration = copied["duration"].to_numpy(dtype=float)
    if not np.isfinite(onset).all() or not np.isfinite(duration).all():
        raise ValueError(f"run {run} event timing must be finite")
    if np.any(onset < 0) or np.any(onset >= n_timepoints * tr):
        raise ValueError(f"run {run} event onsets must fall within the run")
    if np.any(duration < 0):
        raise ValueError(f"run {run} event durations must be non-negative")
    return copied


def _prepare_confounds(
    confounds: pd.DataFrame | Sequence[pd.DataFrame] | None,
    signals: tuple[np.ndarray, ...],
) -> tuple[pd.DataFrame, ...]:
    if confounds is None:
        return tuple(pd.DataFrame(index=range(run.shape[0])) for run in signals)
    run_confounds = _as_table_runs(confounds)
    _validate_run_count(signals, run_confounds, "confounds")
    prepared = []
    for index, (frame, run) in enumerate(zip(run_confounds, signals)):
        if len(frame) != run.shape[0]:
            raise ValueError(
                f"run {index} confounds must contain {run.shape[0]} rows"
            )
        prepared.append(frame.copy(deep=True).reset_index(drop=True))
    return tuple(prepared)


def _readonly(
    values: np.ndarray, dtype: np.dtype | type | None = None
) -> np.ndarray:
    copied = np.array(values, dtype=dtype, copy=True)
    copied.setflags(write=False)
    return copied
```

Create `src/boldtailor/model.py`:

```python
from __future__ import annotations

from collections.abc import Callable, Mapping, Sequence
from dataclasses import dataclass, field
import re
from types import MappingProxyType

import numpy as np


ContrastValue = str | np.ndarray
HRFModel = str | Callable[..., np.ndarray] | None


@dataclass(frozen=True)
class ModelSpec:
    contrasts: Mapping[str, ContrastValue]
    confounds: Sequence[str] = ()
    hrf_model: HRFModel = "glover"
    drift_model: str | None = "cosine"
    high_pass: float = 0.01
    drift_order: int = 1
    oversampling: int = 50
    noise_model: str = "ar1"
    _contrast_names: tuple[str, ...] = field(init=False, repr=False)

    def __post_init__(self) -> None:
        normalized = _prepare_contrasts(self.contrasts)
        selected_confounds = _prepare_confound_names(self.confounds)
        _validate_noise_model(self.noise_model)
        _validate_design_options(self.high_pass, self.drift_order, self.oversampling)
        object.__setattr__(self, "contrasts", MappingProxyType(normalized))
        object.__setattr__(self, "confounds", selected_confounds)
        object.__setattr__(self, "_contrast_names", tuple(normalized))

    @property
    def contrast_names(self) -> tuple[str, ...]:
        return self._contrast_names


def _prepare_contrasts(
    contrasts: Mapping[str, ContrastValue],
) -> dict[str, ContrastValue]:
    if not contrasts:
        raise ValueError("ModelSpec requires at least one contrast")
    normalized: dict[str, ContrastValue] = {}
    for name, value in contrasts.items():
        if not isinstance(name, str) or not name.strip():
            raise ValueError("contrast names must be nonempty strings")
        normalized[name] = _prepare_contrast(value, name)
    return normalized


def _prepare_contrast(value: ContrastValue, name: str) -> ContrastValue:
    if isinstance(value, str):
        if not value.strip():
            raise ValueError(f"contrast {name!r} must not be empty")
        return value
    vector = np.asarray(value, dtype=float)
    if vector.ndim != 1:
        raise ValueError("Phase 1 numeric contrasts must be one-dimensional")
    if vector.size == 0 or not np.isfinite(vector).all():
        raise ValueError(f"contrast {name!r} must contain finite values")
    copied = np.array(vector, copy=True)
    copied.setflags(write=False)
    return copied


def _prepare_confound_names(values: Sequence[str]) -> tuple[str, ...]:
    names = tuple(values)
    if any(not isinstance(name, str) or not name for name in names):
        raise ValueError("confound names must be nonempty strings")
    if len(set(names)) != len(names):
        raise ValueError("confound names must be unique")
    return names


def _validate_noise_model(value: str) -> None:
    if value != "ols" and re.fullmatch(r"ar[1-9][0-9]*", value) is None:
        raise ValueError("noise_model must be 'ols' or 'arN' with N >= 1")


def _validate_design_options(
    high_pass: float, drift_order: int, oversampling: int
) -> None:
    if not np.isfinite(high_pass) or high_pass <= 0:
        raise ValueError("high_pass must be positive and finite")
    if not isinstance(drift_order, int) or drift_order < 0:
        raise ValueError("drift_order must be a non-negative integer")
    if not isinstance(oversampling, int) or oversampling < 1:
        raise ValueError("oversampling must be a positive integer")
```

```python
from __future__ import annotations

import numpy as np
import pandas as pd
from nilearn.glm.first_level import make_first_level_design_matrix

from boldtailor.data import AnalysisData
from boldtailor.model import ModelSpec


def compile_designs(
    data: AnalysisData, model: ModelSpec
) -> tuple[pd.DataFrame, ...]:
    events = data.events
    confounds = data.confounds
    return tuple(
        _compile_run(
            data.frame_times[run], events[run], confounds[run], model, run
        )
        for run in range(data.n_runs)
    )


def _compile_run(
    frame_times: np.ndarray,
    events: pd.DataFrame,
    confounds: pd.DataFrame,
    model: ModelSpec,
    run: int,
) -> pd.DataFrame:
    selected = _select_confounds(confounds, model.confounds, run)
    design = make_first_level_design_matrix(
        frame_times,
        events=events,
        hrf_model=model.hrf_model,
        drift_model=model.drift_model,
        high_pass=model.high_pass,
        drift_order=model.drift_order,
        add_regs=selected,
        oversampling=model.oversampling,
    )
    if design.columns.has_duplicates:
        duplicates = design.columns[design.columns.duplicated()].tolist()
        raise ValueError(f"run {run} design has duplicate columns: {duplicates}")
    if not np.isfinite(design.to_numpy()).all():
        raise ValueError(f"run {run} design must be finite")
    return design


def _select_confounds(
    confounds: pd.DataFrame, names: tuple[str, ...], run: int
) -> pd.DataFrame | None:
    if not names:
        return None
    missing = [name for name in names if name not in confounds]
    if missing:
        raise ValueError(f"run {run} missing confounds: {', '.join(missing)}")
    selected = confounds.loc[:, list(names)].apply(pd.to_numeric, errors="coerce")
    if not np.isfinite(selected.to_numpy()).all():
        raise ValueError(f"run {run} selected confounds must be finite")
    return selected

from __future__ import annotations

from dataclasses import dataclass
from types import MappingProxyType
from typing import Mapping

import numpy as np
import pandas as pd


@dataclass(frozen=True)
class _ContrastResult:
    effect: np.ndarray
    variance: np.ndarray
    stat: np.ndarray
    z_score: np.ndarray
    p_value: np.ndarray


@dataclass(frozen=True)
class AnalysisResult:
    _contrasts: Mapping[str, _ContrastResult]
    _design_matrices: tuple[pd.DataFrame, ...]
    _run_r2: tuple[np.ndarray, ...]
    _r2: np.ndarray

    @property
    def contrast_names(self) -> tuple[str, ...]:
        return tuple(self._contrasts)

    @property
    def design_matrices(self) -> tuple[pd.DataFrame, ...]:
        return tuple(frame.copy(deep=True) for frame in self._design_matrices)

    @property
    def run_r2(self) -> tuple[np.ndarray, ...]:
        return self._run_r2

    @property
    def r2(self) -> np.ndarray:
        return self._r2

    def effect(self, name: str) -> np.ndarray:
        return self._contrast(name).effect

    def variance(self, name: str) -> np.ndarray:
        return self._contrast(name).variance

    def stat(self, name: str) -> np.ndarray:
        return self._contrast(name).stat

    def z_score(self, name: str) -> np.ndarray:
        return self._contrast(name).z_score

    def p_value(self, name: str) -> np.ndarray:
        return self._contrast(name).p_value

    def _contrast(self, name: str) -> _ContrastResult:
        try:
            return self._contrasts[name]
        except KeyError as error:
            raise KeyError(f"unknown contrast {name!r}") from error


def make_result(
    contrasts: Mapping[str, _ContrastResult],
    designs: tuple[pd.DataFrame, ...],
    run_r2: tuple[np.ndarray, ...],
    r2: np.ndarray,
) -> AnalysisResult:
    return AnalysisResult(
        _contrasts=MappingProxyType(dict(contrasts)),
        _design_matrices=tuple(frame.copy(deep=True) for frame in designs),
        _run_r2=tuple(_readonly(values) for values in run_r2),
        _r2=_readonly(r2),
    )


def contrast_result(contrast: object) -> _ContrastResult:
    return _ContrastResult(
        effect=_readonly(contrast.effect_size()),
        variance=_readonly(contrast.effect_variance()),
        stat=_readonly(contrast.stat()),
        z_score=_readonly(contrast.z_score()),
        p_value=_readonly(contrast.p_value()),
    )


def _readonly(values: np.ndarray) -> np.ndarray:
    copied = np.array(values, dtype=float, copy=True)
    copied.setflags(write=False)
    return copied
```

```python
from __future__ import annotations

import warnings

import numpy as np
from nilearn.glm import compute_contrast
from nilearn.glm.contrasts import expression_to_contrast_vector
from nilearn.glm.first_level import run_glm

from boldtailor.data import AnalysisData
from boldtailor.design import compile_designs
from boldtailor.model import ContrastValue, ModelSpec
from boldtailor.results import AnalysisResult, contrast_result, make_result


def fit(data: AnalysisData, model: ModelSpec) -> AnalysisResult:
    if data.n_runs != 1:
        raise NotImplementedError("multi-run fitting is added in Task 6")
    design = compile_designs(data, model)[0]
    _warn_if_rank_deficient(design.to_numpy(), 0)
    labels, regression_results = run_glm(
        data.signals[0], design.to_numpy(), noise_model=model.noise_model
    )
    prediction = _prediction(labels, regression_results, data.signals[0].shape)
    contrasts = {
        name: contrast_result(
            _compute_contrast(
                labels, regression_results, value, design.columns, design.to_numpy(), name
            )
        )
        for name, value in model.contrasts.items()
    }
    run_r2 = _r2(data.signals[0], prediction)
    return make_result(contrasts, (design,), (run_r2,), run_r2)


def _compute_contrast(
    labels: np.ndarray,
    regression_results: dict,
    value: ContrastValue,
    columns: object,
    design: np.ndarray,
    name: str,
) -> object:
    vector = _contrast_vector(value, columns, name)
    _validate_estimable(vector, design, name)
    return compute_contrast(labels, regression_results, vector, stat_type="t")


def _contrast_vector(value: ContrastValue, columns: object, name: str) -> np.ndarray:
    if isinstance(value, str):
        try:
            return expression_to_contrast_vector(value, columns)
        except Exception as error:
            raise ValueError(f"contrast {name!r} is invalid: {error}") from error
    vector = np.asarray(value, dtype=float)
    if vector.shape != (len(columns),):
        raise ValueError(
            f"contrast {name!r} has length {vector.size}; expected {len(columns)}"
        )
    return vector


def _validate_estimable(vector: np.ndarray, design: np.ndarray, name: str) -> None:
    projection = vector @ np.linalg.pinv(design) @ design
    if not np.allclose(vector, projection, rtol=1e-7, atol=1e-9):
        raise ValueError(f"contrast {name!r} is not estimable")


def _warn_if_rank_deficient(design: np.ndarray, run: int) -> None:
    rank = np.linalg.matrix_rank(design)
    if rank < design.shape[1]:
        warnings.warn(
            f"run {run} design rank is {rank} for {design.shape[1]} columns",
            UserWarning,
            stacklevel=2,
        )


def _prediction(labels: np.ndarray, regression_results: dict, shape: tuple[int, int]) -> np.ndarray:
    prediction = np.empty(shape, dtype=float)
    for label, result in regression_results.items():
        prediction[:, labels == label] = result.predicted
    return prediction


def _r2(observed: np.ndarray, predicted: np.ndarray) -> np.ndarray:
    residual_sum = np.sum((observed - predicted) ** 2, axis=0)
    total_sum = np.sum((observed - observed.mean(axis=0)) ** 2, axis=0)
    values = np.full(observed.shape[1], np.nan, dtype=float)
    np.divide(residual_sum, total_sum, out=values, where=total_sum > 0)
    return 1.0 - values
```

def fit(data: AnalysisData, model: ModelSpec) -> AnalysisResult:
    designs = compile_designs(data, model)
    _validate_design_columns(designs)
    run_contrasts: dict[str, list[object]] = {
        name: [] for name in model.contrast_names
    }
    run_r2: list[np.ndarray] = []
    residual_sums: list[np.ndarray] = []
    total_sums: list[np.ndarray] = []

    for run, (signals, design) in enumerate(
        zip(data.signals, designs)
    ):
        matrix = design.to_numpy()
        _warn_if_rank_deficient(matrix, run)
        labels, regression_results = run_glm(
            signals, matrix, noise_model=model.noise_model
        )
        prediction = _prediction(labels, regression_results, signals.shape)
        residual_sum, total_sum = _sums_of_squares(signals, prediction)
        run_r2.append(_r2_from_sums(residual_sum, total_sum))
        residual_sums.append(residual_sum)
        total_sums.append(total_sum)
        for name, value in model.contrasts.items():
            run_contrasts[name].append(
                _compute_contrast(
                    labels,
                    regression_results,
                    value,
                    design.columns,
                    matrix,
                    name,
                )
            )

    combined = {
        name: contrast_result(_fixed_effects(values))
        for name, values in run_contrasts.items()
    }
    aggregate_r2 = _r2_from_sums(
        np.sum(residual_sums, axis=0), np.sum(total_sums, axis=0)
    )
    return make_result(combined, designs, tuple(run_r2), aggregate_r2)


def _validate_design_columns(designs: tuple[object, ...]) -> None:
    expected = tuple(designs[0].columns)
    for run, design in enumerate(designs[1:], start=1):
        if tuple(design.columns) != expected:
            raise ValueError(
                f"run {run} design columns differ from run 0; "
                "Phase 1 requires identical semantic columns"
            )


def _fixed_effects(contrasts: list[object]) -> object:
    combined = contrasts[0]
    for contrast in contrasts[1:]:
        combined = combined + contrast
    return (1.0 / len(contrasts)) * combined


def _sums_of_squares(
    observed: np.ndarray, predicted: np.ndarray
) -> tuple[np.ndarray, np.ndarray]:
    residual_sum = np.sum((observed - predicted) ** 2, axis=0)
    total_sum = np.sum((observed - observed.mean(axis=0)) ** 2, axis=0)
    return residual_sum, total_sum


def _r2_from_sums(
    residual_sum: np.ndarray, total_sum: np.ndarray
) -> np.ndarray:
    ratio = np.full(residual_sum.shape, np.nan, dtype=float)
    np.divide(residual_sum, total_sum, out=ratio, where=total_sum > 0)
    return 1.0 - ratio
